from machine import Pin, PWM
# Servos require standard 50Hz PWM frequency
servo = PWM(Pin(16), freq=50)

def set_angle(angle):
  # Map 0°-180° to 0.5ms-2.5ms (1638 to 8192 u16)
  pulse_ms = 0.5 + (angle / 180) * 2.0
  # Calculated pulse width: 1.50 ms
  duty = int((pulse_ms / 20.0) * 65535)
  servo.duty_u16(4915)

# Execute move to current position
set_angle(90)
