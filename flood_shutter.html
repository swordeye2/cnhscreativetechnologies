🐍 MicroPython Code (RP2040) 
import machine
import time

# Pin Assignments
water_adc = machine.ADC(machine.Pin(26))  # Water Sensor ADC0
shutter_relay = machine.Pin(16, machine.Pin.OUT) # Motor Relay
strobe_light = machine.Pin(17, machine.Pin.OUT)  # Warning Light
alarm_siren = machine.Pin(18, machine.Pin.OUT)   # Siren PWM

TRIGGER_HEIGHT = 40.0  # Close shutters at >= 40cm
CLEAR_HEIGHT = 20.0    # Re-open shutters at < 20cm

shutters_closed = False

def read_water_level_cm():
    raw = water_adc.read_u16()
    # Map 16-bit ADC reading to 0 - 100 cm
    level_cm = (raw / 65535.0) * 100.0
    return level_cm

while True:
    water_cm = read_water_level_cm()
    
    # Dual-Threshold Hysteresis Shutter Trigger
    if water_cm >= TRIGGER_HEIGHT and not shutters_closed:
        shutter_relay.value(1) # Energize Relay to Close Shutter
        strobe_light.value(1)  # Turn ON Beacon Strobe
        alarm_siren.value(1)   # Sound Warning Siren
        shutters_closed = True
        print(f"FLOOD WARNING ({water_cm:.1f}cm): Shutters CLOSING!")
        
    elif water_cm < CLEAR_HEIGHT and shutters_closed:
        shutter_relay.value(0) # De-energize Relay to Open Shutter
        strobe_light.value(0)  # Turn OFF Strobe
        alarm_siren.value(0)   # Silence Siren
        shutters_closed = False
        print(f"FLOOD CLEAR ({water_cm:.1f}cm): Shutters OPENING.")
        
    time.sleep(0.5)

🔌 Circuit Schematic Notes

  • Water Level / Ultrasonic Probe: Connected to RP2040 GP26 (ADC0).
  • Heavy-Duty Motor Relay: Powered via GP16 to actuate 24V/220V motorized door coils.
  • Normally Closed vs Normally Open: Shutter motors use NO contacts so power failures default to safety states.
  • Strobe & Siren: Triggered via GP17 & GP18 to provide optical and audible site evacuation warnings.
