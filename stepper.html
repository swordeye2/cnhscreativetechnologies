Stepper Motor Simulator

4-Phase Full-Step Sequence, Coil Excitation Matrix & MicroPython Inspector

Step Index: 0 / 3
Active Sequence Row: [1, 1, 0, 0]
Rotor Angle: 22.5°
GP2 (IN1): 1 GP3 (IN2): 1 GP4 (IN3): 0 GP5 (IN4): 0

import machine, utime

# Define GPIO outputs for IN1, IN2, IN3, IN4
pins = [
  machine.Pin(2, machine.Pin.OUT), # pins[0] -> IN1
  machine.Pin(3, machine.Pin.OUT), # pins[1] -> IN2
  machine.Pin(4, machine.Pin.OUT), # pins[2] -> IN3
  machine.Pin(5, machine.Pin.OUT) # pins[3] -> IN4
]

# Full-Step 2-Coil Excitation Sequence Matrix
seq = [
  [1, 1, 0, 0], # Step 0: IN1+IN2 active (3.3V)
  [0, 1, 1, 0], # Step 1: IN2+IN3 active (3.3V)
  [0, 0, 1, 1], # Step 2: IN3+IN4 active (3.3V)
  [1, 0, 0, 1] # Step 3: IN4+IN1 active (3.3V)
]

# Loop through sequence rows
for step in seq:
  # 'step' is current row array: [1, 1, 0, 0]
  for i in range(4):
    # pins[i] selects pin, step[i] gets 0 or 1 value
    pins[i].value(step[i]) # Sets HIGH (1) or LOW (0)

  # Delay between steps (controls rotation speed)
  utime.sleep_ms(500)
