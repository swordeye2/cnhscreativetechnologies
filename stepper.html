import machine, utime
# GPIO Pin setup (Reversed order flips direction)
pins = [GP2, GP3, GP4, GP5]
# 4-Phase Full Step Matrix
seq = [
  [1, 1, 0, 0], # Step 0: IN1+IN2
  [0, 1, 1, 0], # Step 1: IN2+IN3
  [0, 0, 1, 1], # Step 2: IN3+IN4
  [1, 0, 0, 1] # Step 3: IN4+IN1
]
for step in seq:
  for i in range(4):
    pins[i].value(step[i])
  utime.sleep_ms(500)
