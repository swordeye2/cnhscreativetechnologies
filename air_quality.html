🐍 MicroPython Code (RP2040) 
import machine
import time

# Pin Definitions
adc_pin = machine.ADC(machine.Pin(26)) # MQ Gas Sensor
relay_pin = machine.Pin(16, machine.Pin.OUT) # Exhaust Fan

THRESHOLD_PPM = 800
HYSTERESIS_PPM = 100

def read_ppm():
    raw = adc_pin.read_u16()
    voltage = (raw / 65535.0) * 3.3
    # Map voltage to approximate PPM scale
    ppm = int((voltage / 3.3) * 2000)
    return ppm, raw

relay_active = False

while True:
    ppm, raw = read_ppm()
    
    # Dual-threshold Hysteresis Relay Control
    if ppm >= THRESHOLD_PPM and not relay_active:
        relay_pin.value(1) # Turn ON Exhaust Fan
        relay_active = True
        print(f"ALERT: Gas {ppm} PPM! Fan ON.")
    
    elif ppm < (THRESHOLD_PPM - HYSTERESIS_PPM) and relay_active:
        relay_pin.value(0) # Turn OFF Exhaust Fan
        relay_active = False
        print(f"CLEAR: Gas {ppm} PPM. Fan OFF.")
        
    time.sleep(0.5)

🔌 Circuit Schematic Notes

  • MQ-135 Sensor: VCC → 5V, GND → GND, AO → RP2040 GP26 (ADC0).
  • Relay Module: VCC → 5V, GND → GND, IN → RP2040 GP16.
  • Exhaust Fan: Powered via Relay NO (Normally Open) contact switch circuit.
  • Hysteresis Logic: Prevents rapid relay contact chatter when air quality fluctuates around the target threshold.
